<script>
const LS = { set:(k,v)=>localStorage.setItem('onedish_'+k,JSON.stringify(v)),
             get:(k,d)=>{const v=localStorage.getItem('onedish_'+k);return v?JSON.parse(v):d;} };

const DEFAULTS = {
  name:'One dish. Lk', title:'One dish. Lk',
  kicker:'Homemade • Fresh • Sri Lankan',
  sub:'Discover delicious home-cooked dishes, share your favourites, and support the cooks you love.',
  cur:'Rs', color:'#e0782f', hero: null
};

let settings = Object.assign({}, DEFAULTS, LS.get('settings', {}));
let dishes   = LS.get('dishes', []);
let txns     = LS.get('txns', []);
let comments = LS.get('comments', {});
let selAmt = 100, tmpDishImg = null;

const fmt = n => settings.cur + ' ' + Number(n||0).toLocaleString();
const $ = id => document.getElementById(id);
const go = id => document.getElementById(id).scrollIntoView({behavior:'smooth'});

function toast(msg){ const t=$('toast'); t.textContent=msg; t.classList.add('show');
  setTimeout(()=>t.classList.remove('show'),2200); }

/* ---------- APPLY SETTINGS ---------- */
function applySettings(){
  document.documentElement.style.setProperty('--primary', settings.color);
  document.documentElement.style.setProperty('--primary-dark', shade(settings.color,-25));
  $('brandName').textContent = settings.name;
  document.title = settings.name + ' — Taste of Home';
  $('heroTitle').textContent = settings.title;
  $('heroSub').textContent = settings.sub;
  $('heroKicker').textContent = settings.kicker;
  $('footName').textContent = settings.name;
  document.querySelectorAll('.curSym').forEach(e=>e.textContent=settings.cur);
  $('heroBg').style.backgroundImage = settings.hero
    ? `url(${settings.hero})`
    : 'linear-gradient(135deg,#e0782f,#a0451a), url("https://images.unsplash.com/photo-1504674900247-0877df9cc836?w=1600&q=80")';
  if(!settings.hero) $('heroBg').style.background = 'url("https://images.unsplash.com/photo-1504674900247-0877df9cc836?w=1600&q=80") center/cover';
  renderDishes(); renderBalance(); renderTxns(); buildChips();
}
function shade(hex,p){const n=parseInt(hex.slice(1),16),a=Math.round(2.55*p);
  return '#'+(0x1000000+(Math.min(255,Math.max(0,(n>>16)+a))<<16)+(Math.min(255,Math.max(0,((n>>8)&255)+a))<<8)+Math.min(255,Math.max(0,(n&255)+a))).toString(16).slice(1);}

/* ---------- DISHES ---------- */
function renderDishes(){
  const g=$('dishGrid'); g.innerHTML='';
  $('dishEmpty').style.display = dishes.length?'none':'block';
  dishes.forEach(d=>{
    const c = (comments[d.id]||[]).length;
    g.innerHTML += `<div class="dish" onclick="openDish(${d.id})">
      <button class="del" onclick="event.stopPropagation();delDish(${d.id})">✕</button>
      <img src="${d.img||'https://images.unsplash.com/photo-1546069901-ba9599a7e63c?w=600&q=80'}" alt="">
      <div class="body">
        <h3>${esc(d.name)}</h3>
        <div class="price">${d.price?fmt(d.price):'—'}</div>
        <div class="desc">${esc(d.desc||'')}</div>
        <div class="meta">💬 ${c} comment${c===1?'':'s'} • 🕒 ${d.when}</div>
      </div></div>`;
  });
}
function esc(s){return String(s).replace(/[&<>"]/g,m=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[m]));}

function openAdd(){ $('addModal').classList.add('open'); }
function closeModal(id){ $(id).classList.remove('open'); }
document.querySelectorAll('.modal-bg').forEach(m=>m.addEventListener('click',e=>{if(e.target===m)m.classList.remove('open');}));

function previewImg(input, boxId){
  const f=input.files[0]; if(!f)return;
  const r=new FileReader();
  r.onload=e=>{ tmpDishImg=e.target.result; $(boxId).innerHTML=`<img src="${tmpDishImg}">`; };
  r.readAsDataURL(f);
}
function saveDish(){
  const name=$('dishName').value.trim();
  if(!name){toast('Please give the dish a name ✍️');return;}
  dishes.unshift({ id:Date.now(), name, desc:$('dishDesc').value.trim(),
    price:$('dishPrice').value||'', img:tmpDishImg,
    when:new Date().toLocaleDateString('en-GB',{day:'numeric',month:'short'}) });
  LS.set('dishes',dishes);
  $('dishName').value='';$('dishDesc').value='';$('dishPrice').value='';tmpDishImg=null;$('dishPreview').innerHTML='';
  closeModal('addModal'); renderDishes(); toast('Dish published! 🎉');
}
function delDish(id){
  if(!confirm('Delete this dish?'))return;
  dishes=dishes.filter(d=>d.id!==id); delete comments[id];
  LS.set('dishes',dishes); LS.set('comments',comments); renderDishes(); toast('Dish deleted');
}

/* ---------- DISH VIEW + COMMENTS ---------- */
let curDish=null;
function openDish(id){
  curDish=id;
  const d=dishes.find(x=>x.id===id); if(!d)return;
  const list=comments[id]||[];
  $('viewModalBody').innerHTML=`
    <button class="close" onclick="closeModal('viewModal')">✕</button>
    <img src="${d.img||'https://images.unsplash.com/photo-1546069901-ba9599a7e63c?w=800&q=80'}">
    <h2 style="margin-top:14px">${esc(d.name)} ${d.price?`<span style="color:var(--primary)">${fmt(d.price)}</span>`:''}</h2>
    <p style="color:#6d6155;margin:6px 0 4px">${esc(d.desc||'')}</p>
    <h3 style="margin-top:18px">Comments (${list.length}) 💬</h3>
    <div id="cList">${list.map(c=>commentHTML(c)).join('')||'<p class="hint">No comments yet — be the first!</p>'}</div>
    <input id="cName" placeholder="Your name" style="width:100%;padding:11px 14px;border-radius:12px;border:1.5px solid #e8dccd;margin-top:12px;outline:none">
    <textarea id="cText" placeholder="Write a comment..."></textarea>
    <button class="btn btn-primary" style="width:100%" onclick="postComment()">Post Comment 💬</button>`;
  $('viewModal').classList.add('open');
}
function commentHTML(c){
  return `<div class="comment"><div class="avatar">${esc(c.name[0]||'?').toUpperCase()}</div>
    <div><div class="c-name">${esc(c.name)} <span class="c-time">• ${c.when}</span></div><p>${esc(c.text)}</p></div></div>`;
}
function postComment(){
  const name=$('cName').value.trim()||'Anonymous';
  const text=$('cText').value.trim();
  if(!text){toast('Write something first ✍️');return;}
  (comments[curDish]=comments[curDish]||[]).push({name,text,
    when:new Date().toLocaleString('en-GB',{day:'numeric',month:'short',hour:'2-digit',minute:'2-digit'})});
  LS.set('comments',comments);
  openDish(curDish); renderDishes(); toast('Comment posted! 💬');
}

/* ---------- MONEY ---------- */
function buildChips(){
  $('chipRow').innerHTML=[100,250,500,1000].map(a=>
    `<div class="chip ${a===selAmt?'sel':''}" onclick="selAmt=${a};buildChips()">${fmt(a)}</div>`).join('');
}
function addMoney(){
  const amt = parseFloat($('customAmt').value) || selAmt;
  if(amt<=0){toast('Enter a valid amount');return;}
  const donor=$('donorName').value.trim()||'Anonymous supporter';
  txns.unshift({donor, amt, when:new Date().toLocaleString('en-GB',{day:'numeric',month:'short',hour:'2-digit',minute:'2-digit'})});
  LS.set('txns',txns);
  $('customAmt').value='';$('donorName').value='';
  renderBalance(); renderTxns(); toast('Thank you for the tip! 💛');
}
function renderBalance(){
  const total=txns.reduce((s,t)=>s+t.amt,0);
  $('bigBalance').textContent=fmt(total);
